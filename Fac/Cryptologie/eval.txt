    def chiffre(self,m):
        mC=Message([])
        #On doit mémoriser la longueur totale de la liste à chiffrer
        mC.ajouteLongueValeur(len(m))
        pos=0
        while pos<len(m):
            x,pos=m.lisMot64Bits(pos)
            mC.ajouteMot64Bits( self.f(x) )
        return mC

    def dechiffre(self,mC):
        mD=Message([])
        longueur,pos=mC.lisLongueValeur(0)

        while pos<len(mC):
            y,pos=mC.lisMot64Bits(pos)
            mD.ajouteMot64Bits( self.f.valInv(y) )
        return Message(mD[:longueur])

    def chiffrer(self, message):
        message_chiffre = []
        blp= self.iv
        for octet in message:
            bloc = octet ^ bp
            bc = self.bijection(bloc)
            message_chiffre.append(bc)
            bp = bc
        return message_chiffre
    def dechiffrer(self, message_chiffre):
        message = []
        bp = self.iv
        for bc in message_chiffre:
            bloc = self.bijection.valInv(bc)
            octet = bloc ^ bp            
            message.append(octet)
bp = bc
        return message

Feistel:
 def __init__(self, nbTours=4, graine=0):
        self.nbTours = nbTours
        random.seed(graine)
        self.cles = [random.randint(0, 255) for _ in range(nbTours)]

    def __repr__(self):
        return f"FBijFeistel(nbTours={self.nbTours})"

    def __call__(self, octet):
        gauche = octet >> 4
        droite = octet & 0x0F
        for cle in self.cles:
            nouvelleGauche = droite
            droite = gauche ^ self.fonctionDeTour(droite, cle)
            gauche = nouvelleGauche
        return (gauche << 4) | droite
