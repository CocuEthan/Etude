# nous reprenons les fichier de config de la v1

# Étape 1 : Préparation du dossier v2
#  Nous allons récupérer tout ce qui fonctionnait dans la v1 
# (le Dockerfile, les binaires, les libs) et les copier dans la v2.

# Exécutez ces commandes sur le manager :

Bash

# 1. Aller dans le dossier v2
cd ~/projet/v2

# 2. Copier les fichiers binaires et bibliothèques depuis le dossier parent (ou v1)
cp ../v1/geonet .
cp ../v1/buscom .
cp ../v1/libasnd.so .
cp ../v1/libbuscomd.so .

# 3. Copier le Dockerfile corrigé de la v1
cp ../v1/Dockerfile .

# 4. Copier le dossier de configuration de la v1 
cp -r ../v1/config .
# Créer le dossier logs
mkdir logs
# Étape 2 : Création du fichier compose.yml

# Bash
nano compose.yml
# Collez ce contenu :

YAML
services:
  buscom:
    build: .
    container_name: buscom_v2
    user: utilisateur
    command: /usr/bin/buscom
    volumes:
      - ./config:/etc/its/config
    expose:
      - "2775"
    tty: true

  geonet:
    build: .
    container_name: geonet_v2
    privileged: true
    depends_on:
      - buscom
    command: sh -c "rm -f /tmp/geonet.lock && /usr/sbin/geonet -fnmlt"
    volumes:
      - ./config:/etc/its/config
      - ./logs:/var/log/its
#  Sauvegardez et quittez.

# Étape 3 : Modification de la configuration (buscom.json)

# Ouvrez le fichier :

#  Bash

nano config/buscom.json
# Modifiez-le pour ajouter l'adresse cible. 

JSON

{
    "port": 2775,
    "ip": "buscom"
}


# Maintenans nous pouvons lancé le build 
sudo docker compose up --build -d

# afficher le retour de buscom: 
sudo docker compose logs -f buscom

#  arreter le docker proprement: 
sudo docker compose down
