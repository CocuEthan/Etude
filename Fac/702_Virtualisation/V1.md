# V1
# On commence par vérifier La présence ou non des fichiers initiaux nécessaire :
cd ~/projet/v1 
ls -l

# On crée le fichier docker:
nano Dockerfile

# On met ceci dans le docker file:

FROM archlinux:latest

RUN pacman -Sy --noconfirm \
    crypto++ \
    gpsd \
    libpcap \
    qt6-base \
    qt6-connectivity \
    qt6-positioning \
    tcpdump
RUN ln -s $(ls /usr/lib/libgps.so.* | head -n 1) /usr/lib/libgps.so.31
RUN ln -s $(ls /usr/lib/libicui18n.so.* | head -n 1) /usr/lib/libicui18n.so.78
RUN ln -s $(ls /usr/lib/libicuuc.so.* | head -n 1) /usr/lib/libicuuc.so.78
RUN ln -s $(ls /usr/lib/libicudata.so.* | head -n 1) /usr/lib/libicudata.so.78
COPY libasnd.so /usr/lib/
COPY libbuscomd.so /usr/lib/

COPY geonet /usr/sbin/
RUN chmod +x /usr/sbin/geonet

COPY buscom /usr/bin/
RUN chmod +x /usr/bin/buscom

RUN useradd -m utilisateur
RUN mkdir -p /etc/its/config /var/log/its/pcap

CMD rm -f /tmp/geonet.lock && /usr/sbin/geonet -fnmlt


#  On crée le dossier config qui contiendra les json:
mkdir -p config
nano config/system.json
#  puis s'assurer que c'est valur sont présente, si elles sont différente les modifier
{
    "interface": "eth0",
    "itsGnSecurity": false
}
nano config/constantes.json
#  puis s'assurer que c'est valur sont présente, si elles sont différente les modifier
{
    "stationId": 1001,
    "mid": 10,
    "logLevel": "debug"
}
nano config/buscom.json
{
    "port": 2775
}


#  On crée le dossier qui contiendra les log:
mkdir -p logs

#  On construit l'image:

Built le docker: sudo docker build -t projet\_v1 . 

# On lance le conteneur:
docker run -it --rm --privileged -v $(pwd)/config:/etc/its/config -v $(pwd)/logs:/var/log/its projet_v1


#  On lance maintenant buscom:
# Pour trouver le nom du conteneur: 
docker ps 
docker exec -it -u utilisateur <NOM_DU_CONTENEUR> /usr/bin/buscom


#  Lancé geonet:
docker run -it --rm --privileged \
  --name conteneur_v1 \
  -v $(pwd)/config:/etc/its/config \
  -v $(pwd)/logs:/var/log/its \
  projet_v1

#  Lancé buscom
docker exec -it -u utilisateur conteneur\_v1 /usr/bin/buscom
